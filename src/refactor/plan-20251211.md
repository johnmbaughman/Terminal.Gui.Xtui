3. **Reduce complexity** - Extract complex logic into focused helper classes

```csharp
internal static class SyntaxHelpers
{
    /// <summary>
    /// Creates an object creation expression with an object initializer.
    /// Example: new Button { Text = "Click Me", Enabled = true }
    /// </summary>
    public static ObjectCreationExpressionSyntax CreateObjectWithInitializer(
        string fullTypeName,
        Dictionary<string, string> attributes)
    {
        string typeName = TypeNameHelpers.GetLocalTypeName(fullTypeName);
        ObjectCreationExpressionSyntax objectCreation = ObjectCreationExpression(IdentifierName(typeName))
            .WithArgumentList(ArgumentList());

        if (attributes.Count > 0)
        {
            IEnumerable<AssignmentExpressionSyntax> assignments = attributes.Select(attr =>
                AssignmentExpression(
                    SyntaxKind.SimpleAssignmentExpression,
                    IdentifierName(attr.Key),
                    ObjectParsingHelpers.ParseValueWithType(attr.Value, attr.Key)));

            InitializerExpressionSyntax initializer = InitializerExpression(
                SyntaxKind.ObjectInitializerExpression,
                SeparatedList<ExpressionSyntax>(assignments));

            objectCreation = objectCreation.WithInitializer(initializer);
        }

        return objectCreation;
    }

    /// <summary>
    /// Creates a local variable declaration: var varName = initializer;
    /// </summary>
    public static LocalDeclarationStatementSyntax CreateLocalDeclaration(
        string variableName,
        ExpressionSyntax initializer)
    {
        return LocalDeclarationStatement(
            VariableDeclaration(IdentifierName("var"))
                .WithVariables(
                    SingletonSeparatedList(
                        VariableDeclarator(Identifier(variableName))
                            .WithInitializer(EqualsValueClause(initializer)))));
    }

    /// <summary>
    /// Creates a private field declaration: private TypeName? fieldName;
    /// </summary>
    public static FieldDeclarationSyntax CreateFieldDeclaration(
        string fieldName,
        string typeName,
        bool isPrivate = true)
    {
        var modifiers = isPrivate 
            ? TokenList(Token(SyntaxKind.PrivateKeyword))
            : TokenList();

        return FieldDeclaration(
            VariableDeclaration(NullableType(IdentifierName(typeName)))
                .WithVariables(SingletonSeparatedList(VariableDeclarator(Identifier(fieldName)))))
            .WithModifiers(modifiers);
    }

    /// <summary>
    /// Creates a this.field = value assignment: this.fieldName = varName;
    /// </summary>
    public static ExpressionStatementSyntax CreateThisAssignment(
        string fieldName,
        string variableName)
    {
        return ExpressionStatement(
            AssignmentExpression(
                SyntaxKind.SimpleAssignmentExpression,
                MemberAccessExpression(
                    SyntaxKind.SimpleMemberAccessExpression,
                    ThisExpression(),
                    IdentifierName(fieldName)),
                IdentifierName(variableName)));
    }

    /// <summary>
    /// Creates a this.field = expression assignment
    /// </summary>
    public static ExpressionStatementSyntax CreateThisPropertyAssignment(
        string propertyName,
        ExpressionSyntax value)
    {
        return ExpressionStatement(
            AssignmentExpression(
                SyntaxKind.SimpleAssignmentExpression,
                MemberAccessExpression(
                    SyntaxKind.SimpleMemberAccessExpression,
                    ThisExpression(),
                    IdentifierName(propertyName)),
                value));
    }

    /// <summary>
    /// Creates: parent.Add(child1, child2, ...);
    /// </summary>
    public static ExpressionStatementSyntax CreateAddInvocation(
        string parentVarName,
        params string[] childVarNames)
    {
        return ExpressionStatement(
            InvocationExpression(
                MemberAccessExpression(
                    SyntaxKind.SimpleMemberAccessExpression,
                    IdentifierName(parentVarName),
                    IdentifierName("Add")))
                .WithArgumentList(
                    ArgumentList(
                        SeparatedList(
                            childVarNames.Select(name => Argument(IdentifierName(name)))))));
    }

    /// <summary>
    /// Creates: this.Add(child);
    /// </summary>
    public static ExpressionStatementSyntax CreateThisAddInvocation(
        params string[] childVarNames)
    {
        return ExpressionStatement(
            InvocationExpression(
                MemberAccessExpression(
                    SyntaxKind.SimpleMemberAccessExpression,
                    ThisExpression(),
                    IdentifierName("Add")))
                .WithArgumentList(
                    ArgumentList(
                        SeparatedList(
                            childVarNames.Select(name => Argument(IdentifierName(name)))))));
    }

    /// <summary>
    /// Creates: this.Add(this.field1, this.field2, ...);
    /// </summary>
    public static ExpressionStatementSyntax CreateThisAddInvocationWithFieldRefs(
        params string[] fieldNames)
    {
        return ExpressionStatement(
            InvocationExpression(
                MemberAccessExpression(
                    SyntaxKind.SimpleMemberAccessExpression,
                    ThisExpression(),
                    IdentifierName("Add")))
                .WithArgumentList(
                    ArgumentList(
                        SeparatedList(
                            fieldNames.Select(name => 
                                Argument(
                                    MemberAccessExpression(
                                        SyntaxKind.SimpleMemberAccessExpression,
                                        ThisExpression(),
                                        IdentifierName(name))))))));
    }

    /// <summary>
    /// Creates a simple member access: left.right
    /// </summary>
    public static MemberAccessExpressionSyntax CreateMemberAccess(
        string leftIdentifier,
        string rightIdentifier)
    {
        return MemberAccessExpression(
            SyntaxKind.SimpleMemberAccessExpression,
            IdentifierName(leftIdentifier),
            IdentifierName(rightIdentifier));
    }

    /// <summary>
    /// Creates a member access chain: obj.prop1.prop2.prop3
    /// </summary>
    public static MemberAccessExpressionSyntax CreateMemberAccessChain(
        params string[] identifiers)
    {
        if (identifiers.Length < 2)
            throw new ArgumentException("Need at least 2 identifiers", nameof(identifiers));

        ExpressionSyntax current = IdentifierName(identifiers[0]);
        for (int i = 1; i < identifiers.Length; i++)
        {
            current = MemberAccessExpression(
                SyntaxKind.SimpleMemberAccessExpression,
                current,
                IdentifierName(identifiers[i]));
        }

        return (MemberAccessExpressionSyntax)current;
    }
}
```

### 1.2 Create TypeNameHelpers.cs

```csharp
internal static class TypeNameHelpers
{
    /// <summary>
    /// Extracts the local type name from a qualified type name.
    /// Example: "Terminal.Gui.Views.Button" -> "Button"
    /// </summary>
    public static string GetLocalTypeName(string qualifiedTypeName)
    {
        if (string.IsNullOrEmpty(qualifiedTypeName))
            return qualifiedTypeName;

        int lastDot = qualifiedTypeName.LastIndexOf('.');
        return lastDot >= 0 ? qualifiedTypeName.Substring(lastDot + 1) : qualifiedTypeName;
    }

    /// <summary>
    /// Gets the namespace portion of a qualified type name.
    /// Example: "Terminal.Gui.Views.Button" -> "Terminal.Gui.Views"
    /// </summary>
    public static string? GetNamespace(string qualifiedTypeName)
    {
        if (string.IsNullOrEmpty(qualifiedTypeName))
            return null;

        int lastDot = qualifiedTypeName.LastIndexOf('.');
        return lastDot >= 0 ? qualifiedTypeName.Substring(0, lastDot) : null;
    }

    /// <summary>
    /// Checks if a type name is qualified (contains namespace).
    /// </summary>
    public static bool IsQualified(string typeName)
    {
        return !string.IsNullOrEmpty(typeName) && typeName.Contains('.');
    }

    /// <summary>
    /// Generates a variable name from a type name and index.
    /// Example: ("Button", 2) -> "button2"
    /// </summary>
    public static string GenerateVarName(string typeName, int index)
    {
        string localName = GetLocalTypeName(typeName);
        return $"{localName.ToLower()}{index}";
    }

    /// <summary>
    /// Gets a control ID or generates a variable name.
    /// </summary>
    public static string GetControlVarName(
        ElementNode node,
        int index)
    {
        if (node.Attributes.TryGetValue("Id", out string? id) && !string.IsNullOrEmpty(id))
            return id;

        string localTypeName = GetLocalTypeName(node.ElementTypeName);
        return GenerateVarName(localTypeName, index);
    }
}
```

```csharp
internal static class NamespaceHelpers
{
    /// <summary>
    /// Recursively collects all C# namespaces from the element tree.
    /// </summary>
    public static HashSet<string> CollectAllNamespaces(ElementNode node)
    {
        var namespaces = new HashSet<string>();

        // Add C# namespaces from current node (filter out XML schema namespaces)
        foreach (var nsUri in node.Namespaces.Values)
        {
            if (!string.IsNullOrEmpty(nsUri) && 
                !nsUri.StartsWith("http://www.w3.org/") && 
                !nsUri.StartsWith("http://schemas.microsoft.com/"))
            {
                string csNamespace = MapXmlNamespaceUriToCSharp(nsUri);
                namespaces.Add(csNamespace);
            }
        }

        // Recursively collect from children
        foreach (var child in node.Children)
        {
            var childNamespaces = CollectAllNamespaces(child);
            foreach (var ns in childNamespaces)
            {
                namespaces.Add(ns);
            }
        }

        return namespaces;
    }

    /// <summary>
    /// Maps an XML namespace URI to a C# namespace.
    /// </summary>
    public static string MapXmlNamespaceUriToCSharp(string uri)
    {
        if (uri == "http://schemas.terminal.gui/xtui")
        {
            return "Terminal.Gui.Views";
        }
        
        // Parse XAML-style clr-namespace declarations
        if (uri.StartsWith("clr-namespace:"))
        {
            string nsDeclaration = uri.Substring("clr-namespace:".Length);
            int assemblyIndex = nsDeclaration.IndexOf(";");
            if (assemblyIndex > 0)
            {
                return nsDeclaration.Substring(0, assemblyIndex);
            }
            return nsDeclaration;
        }
        
        return uri;
    }

    /// <summary>
    /// Builds using directive list from collected namespaces.
    /// </summary>
    public static List<UsingDirectiveSyntax> BuildUsingDirectives(HashSet<string> namespaces)
    {
        var usings = new List<UsingDirectiveSyntax>
        {
            UsingDirective(QualifiedName(
                QualifiedName(IdentifierName("Terminal"), IdentifierName("Gui")),
                IdentifierName("Views"))),
            UsingDirective(QualifiedName(
                QualifiedName(IdentifierName("Terminal"), IdentifierName("Gui")),
                IdentifierName("ViewBase")))
        };

        foreach (var ns in namespaces.Where(ns => 
            !string.IsNullOrEmpty(ns) && ns != "Terminal.Gui.Views"))
        {
            var parts = ns.Split('.');
            NameSyntax qualifiedName = IdentifierName(parts[0]);
            for (int i = 1; i < parts.Length; i++)
            {
                qualifiedName = QualifiedName(qualifiedName, IdentifierName(parts[i]));
            }
            usings.Add(UsingDirective(qualifiedName));
        }

        return usings;
    }
}
```

```csharp
internal static class ChildProcessingHelpers
{
    /// <summary>
    /// Gets children from a container element or direct children of specified type.
    /// </summary>
    public static List<ElementNode> GetContainerOrDirectChildren(
        ElementNode node,
        string containerTypeName,
        string itemTypeName)
    {
        ElementNode? container = node.Children.FirstOrDefault(c => 
            TypeNameHelpers.GetLocalTypeName(c.ElementTypeName) == containerTypeName);

        return container != null 
            ? container.Children 
            : node.Children.Where(c => 
                TypeNameHelpers.GetLocalTypeName(c.ElementTypeName) == itemTypeName).ToList();
    }

    /// <summary>
    /// Processes children and generates Add invocation with all child variables.
    /// </summary>
    public static void ProcessChildrenWithParamArrayAdd(
        ElementNode node,
        string parentVarName,
        IGeneratorFactory generators,
        List<StatementSyntax> statements,
        string? containerType = null,
        string? itemType = null)
    {
        List<string> childVarNames = new List<string>();
        List<ElementNode> itemsToProcess = containerType != null && itemType != null
            ? GetContainerOrDirectChildren(node, containerType, itemType)
            : node.Children;

        for (int i = 0; i < itemsToProcess.Count; i++)
        {
            ElementNode child = itemsToProcess[i];
            string childVarName = TypeNameHelpers.GetControlVarName(child, i);
            
            Generator childGenerator = generators.GetGenerator(child.ElementTypeName);
            StatementSyntax[] childStatements = childGenerator.GenerateStatements(
                child, childVarName, generators);

            statements.AddRange(childStatements);
            childVarNames.Add(childVarName);
        }

        if (childVarNames.Count > 0)
        {
            statements.Add(SyntaxHelpers.CreateAddInvocation(parentVarName, childVarNames.ToArray()));
        }
    }
}
```

```csharp
internal static class ClassGenerationHelpers
{
    /// <summary>
    /// Builds the InitializeComponent method.
    /// </summary>
    public static MethodDeclarationSyntax BuildInitializeComponentMethod(
        List<StatementSyntax> statements)
    {
        return MethodDeclaration(
                PredefinedType(Token(SyntaxKind.VoidKeyword)),
                Identifier("InitializeComponent"))
            .WithModifiers(TokenList(Token(SyntaxKind.PrivateKeyword)))
            .WithBody(Block(statements));
    }

    /// <summary>
    /// Builds a public partial class with specified base type.
    /// </summary>
    public static ClassDeclarationSyntax BuildPartialClass(
        string className,
        string baseTypeName,
        List<MemberDeclarationSyntax> members)
    {
        return ClassDeclaration(className)
            .WithModifiers(TokenList(
                Token(SyntaxKind.PublicKeyword),
                Token(SyntaxKind.PartialKeyword)))
            .WithBaseList(BaseList(SingletonSeparatedList<BaseTypeSyntax>(
                SimpleBaseType(IdentifierName(baseTypeName)))))
            .WithMembers(List(members));
    }

    /// <summary>
    /// Builds the complete compilation unit with namespace, class, and using directives.
    /// </summary>
    public static CompilationUnitSyntax BuildCompilationUnit(
        string namespaceName,
        ClassDeclarationSyntax classDeclaration,
        List<UsingDirectiveSyntax> usings)
    {
        NamespaceDeclarationSyntax namespaceDeclaration = 
            NamespaceDeclaration(IdentifierName(namespaceName))
                .WithMembers(SingletonList<MemberDeclarationSyntax>(classDeclaration));

        return CompilationUnit()
            .WithUsings(List(usings))
            .WithMembers(SingletonList<MemberDeclarationSyntax>(namespaceDeclaration))
            .NormalizeWhitespace();
    }

    /// <summary>
    /// Adds #nullable enable directive to source.
    /// </summary>
    public static string AddNullableDirective(string source)
    {
        return "#nullable enable\n" + source;
    }

    /// <summary>
    /// Generates property assignments for 'this' from node attributes.
    /// Skips the Id attribute.
    /// </summary>
    public static void AddPropertyAssignmentsFromAttributes(
        Dictionary<string, string> attributes,
        List<StatementSyntax> statements)
    {
        foreach (var attr in attributes)
        {
            if (attr.Key == "Id")
                continue;

            statements.Add(SyntaxHelpers.CreateThisPropertyAssignment(
                attr.Key,
                ObjectParsingHelpers.ParseValueWithType(attr.Value, attr.Key)));
        }
    }
}
```

```csharp
internal abstract class BaseControlGenerator : Generator
{
    /// <summary>
    /// Gets the control type name (e.g., "Button", "Label").
    /// </summary>
    protected abstract string ControlTypeName { get; }

    /// <summary>
    /// Override to provide custom child processing logic.
    /// Default implementation handles standard Add() pattern.
    /// </summary>
    protected virtual void ProcessChild(
        ElementNode child,
        string childVarName,
        string parentVarName,
        List<StatementSyntax> statements,
        IGeneratorFactory generators)
    {
        Generator childGenerator = generators.GetGenerator(child.ElementTypeName);
        StatementSyntax[] childStatements = childGenerator.GenerateStatements(
            child, childVarName, generators);

        statements.AddRange(childStatements);
        statements.Add(SyntaxHelpers.CreateAddInvocation(parentVarName, childVarName));
    }

    /// <summary>
    /// Template method for generating statements.
    /// </summary>
    public override StatementSyntax[] GenerateStatements(
        ElementNode node,
        string variableName,
        IGeneratorFactory generators)
    {
        // Create object with initializer
        ObjectCreationExpressionSyntax objectCreation = 
            SyntaxHelpers.CreateObjectWithInitializer(ControlTypeName, node.Attributes);

        // Create local declaration
        List<StatementSyntax> statements = new List<StatementSyntax>
        {
            SyntaxHelpers.CreateLocalDeclaration(variableName, objectCreation)
        };

        // Process children
        if (node.Children.Count > 0)
        {
            for (int i = 0; i < node.Children.Count; i++)
            {
                ElementNode child = node.Children[i];
                string childVarName = TypeNameHelpers.GetControlVarName(child, i);
                ProcessChild(child, childVarName, variableName, statements, generators);
            }
        }

        return statements.ToArray();
    }
}
```

```csharp
internal abstract class BaseContainerGenerator : Generator
{
    /// <summary>
    /// Gets the container type name for the base class.
    /// </summary>
    protected abstract string ContainerTypeName { get; }

    /// <summary>
    /// Override to customize child processing.
    /// </summary>
    protected virtual List<ElementNode> GetChildrenToProcess(ElementNode node)
    {
        return node.Children;
    }

    /// <summary>
    /// Override to add custom statements after children are processed.
    /// </summary>
    protected virtual void AddPostChildStatements(
        List<StatementSyntax> statements,
        List<string> childVarNames)
    {
        // Default: Add all children in one param array call
        if (childVarNames.Count > 0)
        {
            statements.Add(SyntaxHelpers.CreateThisAddInvocation(childVarNames.ToArray()));
        }
    }

    /// <summary>
    /// Template method for generating full class.
    /// </summary>
    public override string GenerateClass(
        ElementNode node,
        string namespaceName,
        string className,
        IGeneratorFactory generators)
    {
        List<StatementSyntax> initializeComponentStatements = new List<StatementSyntax>();
        List<FieldDeclarationSyntax> fieldDeclarations = new List<FieldDeclarationSyntax>();

        // Add property assignments from attributes
        if (node.Attributes.Count > 0)
        {
            ClassGenerationHelpers.AddPropertyAssignmentsFromAttributes(
                node.Attributes,
                initializeComponentStatements);
        }

        // Process children
        List<string> childVarNames = new List<string>();
        List<ElementNode> itemsToProcess = GetChildrenToProcess(node);

        for (int i = 0; i < itemsToProcess.Count; i++)
        {
            ElementNode child = itemsToProcess[i];
            string childVarName = TypeNameHelpers.GetControlVarName(child, i);
            string localTypeName = TypeNameHelpers.GetLocalTypeName(child.ElementTypeName);

            // Generate child statements
            Generator childGenerator = generators.GetGenerator(child.ElementTypeName);
            StatementSyntax[] childStatements = childGenerator.GenerateStatements(
                child, childVarName, generators);
            initializeComponentStatements.AddRange(childStatements);

            // Create field for child
            string fieldName = childVarName;
            fieldDeclarations.Add(SyntaxHelpers.CreateFieldDeclaration(fieldName, localTypeName));

            // Assign to field
            initializeComponentStatements.Add(SyntaxHelpers.CreateThisAssignment(fieldName, childVarName));

            childVarNames.Add(childVarName);
        }

        // Add any post-processing statements
        AddPostChildStatements(initializeComponentStatements, childVarNames);

        // Build method and class
        MethodDeclarationSyntax initMethod = 
            ClassGenerationHelpers.BuildInitializeComponentMethod(initializeComponentStatements);

        List<MemberDeclarationSyntax> members = new List<MemberDeclarationSyntax>();
        members.AddRange(fieldDeclarations);
        members.Add(initMethod);

        ClassDeclarationSyntax classDeclaration = 
            ClassGenerationHelpers.BuildPartialClass(className, ContainerTypeName, members);

        // Collect namespaces and build using directives
        HashSet<string> namespaces = NamespaceHelpers.CollectAllNamespaces(node);
        List<UsingDirectiveSyntax> usings = NamespaceHelpers.BuildUsingDirectives(namespaces);

        // Build compilation unit
        CompilationUnitSyntax compilationUnit = 
            ClassGenerationHelpers.BuildCompilationUnit(namespaceName, classDeclaration, usings);

        return compilationUnit.ToFullString();
    }
}
```
